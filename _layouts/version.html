Use it

{%- if has_bom %}

Import the BOM and declare the modules you need without a version.

<dependencyManagement>
  <dependencies>
    <dependency>
      <groupId>org.wicketstuff</groupId>
      <artifactId>wicketstuff-bom</artifactId>
      <version>{{ r.version }}</version>
      <type>pom</type>
      <scope>import</scope>
    </dependency>
  </dependencies>
</dependencyManagement>
dependencies {
    implementation(platform("org.wicketstuff:wicketstuff-bom:{{ r.version }}"))
}
{%- else %}

{{ r.wicket }}.x has no BOM. Give every Wicket Stuff dependency the version {{ r.version }}:

<dependency>
  <groupId>org.wicketstuff</groupId>
  <artifactId>wicketstuff-select2</artifactId>
  <version>{{ r.version }}</version>
</dependency>
{%- endif %}
